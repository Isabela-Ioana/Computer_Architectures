; Reset Handler

Reset_Handler   PROC
                EXPORT  Reset_Handler             [WEAK]
					
				; program logic
				
				;load addresses (addresses of the beginning )
				LDR r0, =Cards
				LDR r1, =Condition
				LDR r2, =Purchase_price
				LDR r3, =Current_price
				LDR r4, =num_Cards
				
				
				;
				LDR r5,[r4]  ;counter outer loop
				
				MOV r10,#0   ; sum=0
				MOV r11,#0   ; nb_cards
				
				
Outer_Loop
			LDR r6,[r2]        ;first id of puchase_price
			MOV r12, #7        ;contor inner loop
			LDR R3,=Current_price
			
Inner_Loop
				
				LDR r8,[r3]
				
				
				CMP R6,R8
				BEQ Verify_Prices
				
				ADDS R3, R3,#8
				
				SUBS r12,r12,#1
				BNE Inner_Loop
				
			
			
			
			SUBS R5,R5,#1
			BNE Outer_Loop
				
				
Verify_Prices
	LDR r7, [r2,#4]     ;price of the first id
	LDR r8, [r3,#4]     ;price of the first id
	
	CMP r7,r8
	BLT Make_difference
	ADDS R2, R2,#8
	
	B Outer_Loop
	
Make_difference
	;r8-r7
	
	SUBS r8, r8,r7 
	ADDS r10,r10,r8
	ADDS r11,r11,#1
	ADDS R2, R2,#8
	
	
	B Outer_Loop
	
	
				
				
					
					
					
Stop 			B Stop
				LTORG
				
				ALIGN 1
				SPACE 4096  ;

Cards			DCD 0x134, 3, 275, 0x2B9, 0xDC, 151, 2087
Condition		DCD 2087, 2, 275, 0x0, 308, 0x1, 0xDC, 2, 151, 2, 0x3, 0, 697, 2
Purchase_price 	DCD 0x3, 2000, 0x113, 2, 151, 9, 0x134, 45, 2087, 17, 220, 5, 697, 350
Current_price	DCD	0xDC, 3, 151, 16, 3, 3300, 697, 420, 308, 63, 275, 1, 0x827, 3
num_Cards		DCB	7


				ALIGN 1
				SPACE 4096
				ALIGN 2
				
				
				
			   
                ENDP
